    <script>
        const menuButton = document.getElementById("menuButton");
        const navLinks = document.getElementById("navLinks");
        const codeModal = document.getElementById("codeModal");
        const codeTitle = document.getElementById("codeTitle");
        const codeContent = document.getElementById("codeContent");
        const codeClose = document.getElementById("codeClose");
        const logo = document.getElementById("logo");

        function openMenu() {
            navLinks.classList.add("active");
            menuButton.setAttribute("aria-expanded", "true");
            menuButton.setAttribute(
                "aria-label",
                "Close navigation menu"
            );
        }

        function closeMenu() {
            navLinks.classList.remove("active");
            menuButton.setAttribute("aria-expanded", "false");
            menuButton.setAttribute(
                "aria-label",
                "Open navigation menu"
            );
        }

        menuButton.addEventListener("click", function () {
            const menuIsOpen = navLinks.classList.contains("active");

            if (menuIsOpen) {
                closeMenu();
            } else {
                openMenu();
            }
        });

        function getCssCode() {
            const style = document.querySelector("style");

            if (style === null) {
                return "CSS code unavailable.";
            }

            return style.textContent.trim();
        }

        function getHtmlCode() {
            const page = document.documentElement.cloneNode(true);

            const style = page.querySelector("style");
            const script = page.querySelector("script");
            const modal = page.querySelector("#codeModal");

            if (style !== null) {
                style.remove();
            }

            if (script !== null) {
                script.remove();
            }

            if (modal !== null) {
                modal.remove();
            }

            return "<!DOCTYPE html>\n" + page.outerHTML;
        }

        function getJavascriptCode() {
            const script = document.querySelector("script");

            if (script === null) {
                return "JavaScript code unavailable.";
            }

            return script.textContent.trim();
        }

        function openCodeViewer(type) {
            if (type === "css") {
                codeTitle.textContent = "CSS";
                codeContent.textContent = getCssCode();
            }

            if (type === "html") {
                codeTitle.textContent = "HTML";
                codeContent.textContent = getHtmlCode();
            }

            if (type === "javascript") {
                codeTitle.textContent = "JavaScript";
                codeContent.textContent = getJavascriptCode();
            }

            codeModal.classList.add("active");
            codeModal.setAttribute("aria-hidden", "false");
            document.body.classList.add("no-scroll");

            closeMenu();
        }

        function closeCodeViewer() {
            codeModal.classList.remove("active");
            codeModal.setAttribute("aria-hidden", "true");
            document.body.classList.remove("no-scroll");
        }

        const codeLinks = document.querySelectorAll(
            ".nav-links a[data-code]"
        );

        codeLinks.forEach(function (link) {
            link.addEventListener("click", function (event) {
                event.preventDefault();

                const type = link.getAttribute("data-code");

                openCodeViewer(type);
            });
        });

        codeClose.addEventListener(
            "click",
            closeCodeViewer
        );

        codeModal.addEventListener("click", function (event) {
            if (event.target === codeModal) {
                closeCodeViewer();
            }
        });

        document.addEventListener("keydown", function (event) {
            if (event.key === "Escape") {
                closeCodeViewer();
                closeMenu();
            }
        });

        logo.addEventListener("click", function (event) {
            event.preventDefault();

            window.scrollTo({
                top: 0,
                behavior: "smooth"
            });
        });

        document.addEventListener("click", function (event) {
            const clickedNavbar =
                event.target.closest(".navbar");

            if (
                !clickedNavbar &&
                navLinks.classList.contains("active")
            ) {
                closeMenu();
            }
        });
    </script>
